async function ctxFor(code: string, role: DistRole = "Owner", branchName?: string): Promise<Ctx> {
  const t = await db.tenant.findUniqueOrThrow({ where: { code } });
  const u = await db.user.findFirstOrThrow({ where: { tenantId: t.id }, orderBy: { createdAt: "asc" } });
  const branches = await db.branch.findMany({ where: { tenantId: t.id }, orderBy: { createdAt: "asc" } });
  const br = branchName ? branches.find((b) => b.name === branchName)! : branches[0];
  return { tenantId: t.id, tenantName: t.name, userId: u.id, userName: u.name, userEmail: u.email ?? "", realRole: role, role, perms: [...DIST_GRANTS[role]],
    branchId: br.id, branchName: br.name, branchCode: br.code, branches: branches.map((b) => ({ id: b.id, name: b.name, code: b.code })), lang: "en", sessionId: "seed", counterId: null, counterName: null };
}

const C_GST = (i: number, state: string) => {
  const C = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ";
  const base = `${state}DEMOC${String(1000 + i).padStart(4, "0")}D1Z`;
  let s = 0; for (let k = 0; k < 14; k++) { const v = C.indexOf(base[k]) * (k % 2 ? 2 : 1); s += Math.floor(v / 36) + (v % 36); }
  return base + C[(36 - (s % 36)) % 36];
};

    { medicineId: P("Diclomax SR"), batchNo: "DICL0903N", expiry: "2027-01-31", qty: 60, rate: "42", mrp: "72" },
  ] } as never);

  // ── 60 days of trade ──
  const days: string[] = []; for (let d = START; d < TODAY; d = addDays(d, 1)) days.push(d);
  const invoicesBy: Record<string, string[]> = {};
  let dayNo = 0;
  const retry: string[] = [];
  for (const day of days) {
    dayNo++;
    const dow = new Date(day + "T12:00:00Z").getUTCDay();
    if (dow === 0) continue; // Sundays closed
    setClock(day, "10:05:00");
    ctx = await ctxFor(CODE);
    // weekly restock (Mondays) from two suppliers
    if (dow === 1 && dayNo > 3) {
      for (const si of [0, 1, 2, 3]) {
        const stock = await db.batch.groupBy({ by: ["medicineId"], where: { tenantId: ctx.tenantId, branchId: ctx.branchId }, _sum: { qtyAvailable: true } });
        const list = bySupplier(si).filter(({ id, p }) => (stock.find((x) => x.medicineId === id)?._sum.qtyAvailable ?? 0) < p[12] * 2.2);
        if (list.length) await postGrnTx(ctx, { supplierId: supplierIds[si], invoiceNo: `DEMO-SI-${String(inv++).padStart(4, "0")}`, invoiceDate: day, acceptShortExpiry: true, lines: grnLines(list, day, 0.8, 2) } as never).catch((e) => console.log("grn", day, e.message));
      }
    }
    // invoices: 4–9 a day, mostly through orders
    const n = between(6, 12);
    for (let k = 0; k < n; k++) {
      setClock(day, `${String(10 + Math.floor(k * 0.8)).padStart(2, "0")}:${String(between(0, 59)).padStart(2, "0")}:00`);
      const ci = pick([0, 1, 1, 2, 3, 3, 4, 5, 5, 6, 7, 7, 8, 9, 10, 10, 10, 11, 12, 13, 14, 15, 15, 16, 16, 17, 18, 19, 20, 21, 21, 22, 23]);
      if (ci === 0) continue; // keep Shree Ram near its limit for the credit demo
      const items = Array.from(new Set(Array.from({ length: between(2, 6) }, () => between(0, prod.length - 1))))
        .filter((x) => PRODUCTS[x][11] !== "H1" || rnd() < 0.2)
        .map((x) => ({ medicineId: prod[x].id, qty: x === 0 ? pick([20, 30, 50, 100]) : between(5, 30) * (PRODUCTS[x][12] > 200 ? 2 : 1) }));
      if (!items.length) continue;
      try {
        let invoiceId: string;
        if (rnd() < 0.55) {
          const o = await placeOrder({ kind: "user", ctx }, { customerId: chem[ci], source: pick(["salesman", "phone", "office", "salesman"]) as never, idempotencyKey: key(), lines: items });
          await approveOrder(ctx, o.orderId, { allowPartial: true, overrideReason: "Regular customer, payment promised (demo)" });
          const pl = await db.distPickList.findFirstOrThrow({ where: { orderId: o.orderId }, include: { lines: true } });
          await startPicking(ctx, pl.id);
          for (const l of pl.lines) await confirmPickLine(ctx, l.id, l.units);
          await packOrder(ctx, pl.id, between(1, 3));
          invoiceId = (await invoiceOrder(ctx, o.orderId)).invoiceId;
        } else {
          invoiceId = (await directInvoice(ctx, { customerId: chem[ci], idempotencyKey: key(), lines: items, overrideReason: "Regular customer (demo)" })).invoiceId;
        }
        (invoicesBy[day] ??= []).push(invoiceId);
      } catch (e) { if (!/credit|stock|short|licen/i.test((e as Error).message)) console.log("sale", day, (e as Error).message); }
    }
    // dispatch the day's invoices in the afternoon; deliveries the same evening
    const todays = invoicesBy[day] ?? [];
    const trip = [...todays, ...retry.splice(0)];
    if (trip.length) {
      setClock(day, "15:30:00");
      try {
        const d = await createDispatch(ctx, { vehicleNo: pick(["JH01DM1234", "JH01DM5678"]), driverName: pick(["Demo Driver Manoj", "Demo Driver Salim"]), route: pick(["Ranchi city", "Kanke–Ratu", "Doranda–HEC", "Ramgarh"]), invoiceIds: trip } as never);
        await startDispatch(ctx, d.dispatchId);
        setClock(day, "18:10:00");
        const dels = await db.distDelivery.findMany({ where: { dispatchId: d.dispatchId } });
        for (const del of dels) {
          const v = (await invoiceView(ctx.tenantId, del.invoiceId))!;
          const total = Number(v.total ?? 0) / 100;
          const r = rnd();
          if (r < 0.04 && day < addDays(TODAY, -2)) { await recordDelivery(ctx, del.id, { outcome: "failed", note: "Shop closed (demo)" }); retry.push(del.invoiceId); }
          else if (r < 0.35) await recordDelivery(ctx, del.id, { outcome: "delivered", receiverName: "Shop staff (demo)", collected: String(Math.round(total)), mode: rnd() < 0.5 ? "Cash" : "UPI", ref: rnd() < 0.5 ? "" : `UTR${between(100000, 999999)}` });
          else await recordDelivery(ctx, del.id, { outcome: "delivered", receiverName: pick(["Owner (demo)", "Pharmacist (demo)", "Shop staff (demo)"]) });
        }
      } catch (e) { console.log("dispatch", day, (e as Error).message); }
    }
    // office / field collections
    setClock(day, "17:00:00");
    for (let k = 0; k < between(3, 7); k++) {
      const ci = between(1, CHEMISTS.length - 2);
      if ([5, 13, 17, 23, 8].includes(ci) && rnd() < 0.85) continue; // slow payers, for ageing
      const bal = (await customerBalance(db, ctx.tenantId, chem[ci])) / 100;
      if (bal < 2000) continue;
      const amt = Math.round((bal * pick([0.25, 0.35, 0.5, 0.6, 0.8])) / 100) * 100;
      if (amt < 500) continue;
      await collectPayment(ctx, { customerId: chem[ci], amount: String(amt), method: pick(["Cash", "UPI", "UPI", "Bank transfer (NEFT)", "Cheque"]) as never, ref: `REF${between(10000, 99999)}`, chequeNo: String(between(100000, 999999)), bank: "Demo Bank", chequeDate: day } as never).catch(() => {});
    }
    // expenses
    if (dayNo % 7 === 2) await recordExpense(ctx, { account: "Delivery expense", amount: String(between(1800, 3200)), method: "Cash", note: "Van fuel (demo)" }).catch((e) => console.log("exp", e.message));
    if (day.endsWith("-01") || day === START) {
      await recordExpense(ctx, { account: "Rent", amount: "45000", method: "Bank", note: "Godown rent (demo)" }).catch(() => {});
      await recordExpense(ctx, { account: "Salaries", amount: "185000", method: "Bank", note: "Staff salaries (demo)" }).catch(() => {});
      await recordExpense(ctx, { account: "Electricity", amount: "8600", method: "UPI", note: "Godown electricity (demo)" }).catch(() => {});
    }
    // supplier payments fortnightly
    if (dayNo % 14 === 10) for (const si of [0, 1, 2, 3]) await paySupplier({ ...ctx, perms: [...ctx.perms, "payments"] }, { partyId: supplierIds[si], amount: String(pick([150000, 250000, 350000])), method: pick(["Bank transfer (NEFT)", "Cheque"]) as never, ref: `NEFT${between(1000, 9999)}`, chequeNo: String(between(100000, 999999)), bank: "Demo Bank" } as never).catch((e) => console.log("pay", e.message));
    // a few credit notes
    if (dayNo % 9 === 5 && todays[0]) {
      const v = (await invoiceView(ctx.tenantId, todays[0]))!;
      const line = v.lines[0];
      if (line && line.qty >= 2) await createReturn(ctx, { invoiceId: todays[0], reason: pick(["Excess supplied (demo)", "Breakage in transit (demo)", "Near expiry returned by chemist (demo)"]), lines: [{ invoiceLineId: line.id, qty: 1, condition: pick(["saleable", "damaged", "saleable"]) as never, creditPct: 100 }] }).catch((e) => console.log("ret", e.message));
    }
  }

  // ── expiry claims on the principal ──
  setClock(addDays(TODAY, -6), "11:00:00");
  ctx = await ctxFor(CODE);
  const exp = await db.batch.findFirst({ where: { tenantId: t.id, batchNo: "GAST0901X" } });
  if (exp) {
    const c1 = await raiseDistClaim(ctx, { batchId: exp.id, qty: 10, reason: "expiry" }).catch((e) => { console.log("claim", e.message); return null; });
    if (c1) { await acknowledgeClaim(ctx, c1.claimId); setClock(addDays(TODAY, -2), "12:00:00"); await creditClaim(ctx, c1.claimId, { ref: "DEMO-SCN-0042", amount: (c1.expected / 100).toFixed(2), close: true, reason: "Company credit note received (demo)" } as never).catch((e) => console.log("credit", e.message)); }
  }
  const dmg = await db.batch.findFirst({ where: { tenantId: t.id, qtyQuarantined: { gt: 0 } } });
  if (dmg) { setClock(addDays(TODAY, -3), "11:30:00"); await raiseDistClaim(ctx, { batchId: dmg.id, qty: Math.min(1, dmg.qtyQuarantined), reason: "breakage" }).catch((e) => console.log("claim2", e.message)); }

  // ── purchase orders: one placed and received by GRN last week, one placed and open, one draft ──
  setClock(addDays(TODAY, -9), "12:00:00"); ctx = await ctxFor(CODE);
  const po1 = await createPo(ctx, { supplierId: supplierIds[1], lines: [{ medicineId: P("Glucocheck Strips"), qty: 60, rate: "556", mrp: "950" }, { medicineId: P("Telmiva 40"), qty: 200, rate: "91.3", mrp: "156", scheme: "10+1" }], note: "Monthly order (demo)" } as never);
  const po1Id = typeof po1 === "string" ? po1 : (po1 as { id: string }).id;
  await setPoStatus(ctx, po1Id, "place").catch((e) => console.log("po1", e.message));
  setClock(addDays(TODAY, -4), "12:00:00"); ctx = await ctxFor(CODE);
  const po2 = await createPo(ctx, { supplierId: supplierIds[0], lines: [{ medicineId: P("Febrinil 650"), qty: 600, rate: "19.35", mrp: "33", scheme: "10+2" }, { medicineId: P("Amoxirise CV 625"), qty: 150, rate: "133.4", mrp: "228" }], note: "Winter season stock (demo)" } as never);
  await setPoStatus(ctx, typeof po2 === "string" ? po2 : (po2 as { id: string }).id, "place").catch((e) => console.log("po2", e.message));
  await createPo(ctx, { supplierId: supplierIds[3], lines: [{ medicineId: P("Sanitize Plus 500 ml"), qty: 120, rate: "133", mrp: "250" }], note: "Draft: check price first (demo)" } as never).catch((e) => console.log("po3", e.message));

  // ── this morning: orders at every stage for the pipeline ──
  setClock(TODAY, "09:15:00");
  ctx = await ctxFor(CODE);
  const mk = async (ci: number, lines: { medicineId: string; qty: number }[], src = "salesman") => placeOrder({ kind: "user", ctx }, { customerId: chem[ci], source: src as never, idempotencyKey: key(), lines });
  // awaiting approval (two), one of them over its credit limit (held)
  await mk(3, [{ medicineId: P("Azivon 500"), qty: 10 }, { medicineId: P("Pantovista 40"), qty: 12 }]).catch((e) => console.log("o1", e.message));
  await mk(13, [{ medicineId: P("Metforal 500"), qty: 40 }, { medicineId: P("Telmiva 40"), qty: 30 }, { medicineId: P("Atorvex 10"), qty: 20 }, { medicineId: P("Glucocheck Strips"), qty: 5 }], "phone").catch((e) => console.log("o2", e.message));
  await mk(6, [{ medicineId: P("Febrinil 650"), qty: 20 }, { medicineId: P("Cetrinova 10"), qty: 10 }], "portal").catch((e) => console.log("o3", e.message));
  // approved, to pick
  setClock(TODAY, "09:40:00");
  const a1 = await mk(10, [{ medicineId: P("Amoxirise CV 625"), qty: 15 }, { medicineId: P("Montiluk-LC"), qty: 20 }, { medicineId: P("Telmiva 40"), qty: 25 }]);
  await approveOrder(ctx, a1.orderId, { allowPartial: true, overrideReason: "Regular customer, payment promised (demo)" });
  const a2 = await mk(15, [{ medicineId: P("Febrinil 650"), qty: 30 }, { medicineId: P("ORSiva Sachet"), qty: 50 }, { medicineId: P("Calvita-D3"), qty: 10 }]);
  await approveOrder(ctx, a2.orderId, { allowPartial: true, overrideReason: "Regular customer, payment promised (demo)" });
  // being picked
  const a3 = await mk(7, [{ medicineId: P("Cefoxa 200"), qty: 10 }, { medicineId: P("Ranzole-D"), qty: 10 }]);
  await approveOrder(ctx, a3.orderId, { allowPartial: true, overrideReason: "Regular customer, payment promised (demo)" });
  const pl3 = await db.distPickList.findFirstOrThrow({ where: { orderId: a3.orderId } });
  await startPicking(ctx, pl3.id);
  // packed, to invoice
  const a4 = await mk(16, [{ medicineId: P("Thyrovance 50"), qty: 10 }, { medicineId: P("Clopidex 75"), qty: 10 }]);
  await approveOrder(ctx, a4.orderId, { allowPartial: true, overrideReason: "Regular customer, payment promised (demo)" });
  const pl4 = await db.distPickList.findFirstOrThrow({ where: { orderId: a4.orderId }, include: { lines: true } });
  await startPicking(ctx, pl4.id); for (const l of pl4.lines) await confirmPickLine(ctx, l.id, l.units); await packOrder(ctx, pl4.id, 2);
  // a rejected one
  const r1 = await mk(8, [{ medicineId: P("Alprazen 0.25"), qty: 20 }], "phone").catch(() => null);
  if (r1) await rejectOrder(ctx, r1.orderId, "Schedule H1 quantity too high for a rural store; asked for prescription records (demo)").catch((e) => console.log("rej", e.message));
  // invoiced this morning, waiting for dispatch
  setClock(TODAY, "10:20:00");
  for (const ci of [1, 11, 21]) await directInvoice(ctx, { customerId: chem[ci], idempotencyKey: key(), lines: [{ medicineId: P("Pantovista 40"), qty: 10 }, { medicineId: P("Febrinil 650"), qty: 20 }, { medicineId: P("Neurobion-Z Forte"), qty: 6 }], overrideReason: "Regular customer (demo)" }).catch((e) => console.log("inv", e.message));
  // Shree Ram Medical Hall: ₹45,000 outstanding against a ₹50,000 limit (credit-control demo)
  setClock(addDays(TODAY, -8), "11:15:00");
  ctx = await ctxFor(CODE);
  await directInvoice(ctx, { customerId: chem[0], idempotencyKey: key(), lines: [{ medicineId: P("Telmiva 40"), qty: 200 }, { medicineId: P("Atorvex 10"), qty: 200 }, { medicineId: P("Metforal 500"), qty: 150 }], overrideReason: "Festival stock, owner approved (demo)" });
  const b0 = (await customerBalance(db, ctx.tenantId, chem[0])) / 100;
  if (b0 > 45000) await collectPayment(ctx, { customerId: chem[0], amount: (b0 - 45000).toFixed(2), method: "UPI", ref: "UTR-DEMO-4501" } as never);
  setClock(TODAY, "10:40:00");
  ctx = await ctxFor(CODE);
  console.log("Shree Ram balance", (await customerBalance(db, ctx.tenantId, chem[0])) / 100);
  // on hold chemist
  await setChemistStatus(ctx, chem[23], "hold", "Two cheques returned; pay old dues before new supply (demo)");
  console.log("seed done");
});
